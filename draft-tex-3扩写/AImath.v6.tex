% -----------------------------------------------------------------------------
% preferInput（精简版：仅 polished_unified / polished / 基础）
% 优先级：*_polished_unified.tex > *_polished.tex > *.tex
% -----------------------------------------------------------------------------
\makeatletter
\newcommand{\preferInput}[1]{%
  \begingroup
  \edef\prefBase{#1}%
  \edef\prefA{\prefBase\detokenize{_polished_unified_expanded.tex}}%
  \edef\prefB{\prefBase\detokenize{_polished_unified.tex}}%
  \edef\prefC{\prefBase\detokenize{_polished.tex}}%
  \edef\prefD{\prefBase\detokenize{_refined.tex}}%
  \IfFileExists{\prefA}{%
    \typeout{[preferInput] -> Using \prefA}%
    \input{\prefA}%
  }{%
    \IfFileExists{\prefB}{%
      \typeout{[preferInput] -> Using \prefB}%
      \input{\prefB}%
    }{%
      \typeout{[preferInput] -> Fallback to \prefC}%
      \input{\prefC}%
    }%
    {%
      \typeout{[preferInput] -> Fallback to \prefD}%
      \input{\prefD}%
    }%
  }%
  \endgroup
}
\makeatother
